# Use a lightweight Node.js environment
FROM node:22-alpine

# Create a directory inside the container for our app
WORKDIR /app

# Copy package files and install dependencies
COPY package*.json ./
RUN npm install

# Copy the rest of the application code (src/app.js)
COPY . .

# Tell the server which network port to open
EXPOSE 3000

# The command to start the web server
CMD ["npm", "start"]
